# Ferramentas do servidor MCP do Dart (dart mcp-server), uma por linha: nome<TAB>classe.
#
# O servidor sobe com --disable all e um --enable para cada ferramenta das tres
# primeiras classes (projeto/mcp.dart.json). Tudo o que nao estiver habilitado por
# nome fica desligado, inclusive ferramentas que versoes futuras criarem.
#
#   leitura    habilitada; so le. As unicas que o Explore e o revisor recebem.
#   livre      habilitada; consulta ou recarrega o app rodando, sem pedir
#              confirmacao (uso frequente no dia a dia).
#   confirmar  habilitada; chama metodos arbitrarios no app rodando ou muda o
#              alcance do servidor. Tem regra "ask" em settings/global.json.
#   desligada  nome conhecido, fica fora do --enable.
#
# Nomes conferidos no enum ToolNames (lib/src/utils/names.dart do dart-lang/ai) e
# pelo teste testes/conferir-dart.ps1 contra o servidor rodando.
analyze_files	leitura
lsp	leitura
pub_dev_search	leitura
read_package_uris	leitura
rip_grep_packages	leitura
get_runtime_errors	livre
widget_inspector	livre
hot_reload	livre
hot_restart	livre
flutter_driver_command	livre
dtd	livre
vm_service	confirmar
roots	confirmar
pub	desligada
dart_fix	desligada
dart_format	desligada
create_project	desligada
run_tests	desligada
launch_app	desligada
stop_app	desligada
list_devices	desligada
list_running_apps	desligada
get_app_logs	desligada
get_active_location	desligada
